1. Create scripts/test_executable.py:
   ```python
   #!/usr/bin/env python3
   """
   Automated Testing for Packaged TuroArnis Executable
   Tests launch, imports, and basic functionality without requiring camera.
   """
   import subprocess
   import os
   import sys
   import time
   import tempfile
   from pathlib import Path
   
   EXE_PATH = Path("dist/TuroArnis/TuroArnis.exe")
   INTERNAL_PATH = Path("dist/TuroArnis/_internal")
   
   def test_executable_launch():
       """Test that executable launches and shows expected output."""
       print("TEST: Executable Launch")
       if not EXE_PATH.exists():
           print(f"  FAIL: {EXE_PATH} not found")
           return False
       
       # Launch with timeout to check it starts
       try:
           proc = subprocess.Popen(
               [str(EXE_PATH)],
               stdout=subprocess.PIPE,
               stderr=subprocess.PIPE,
               text=True
           )
           # Wait briefly for startup
           time.sleep(3)
           
           # Check process is still running (no immediate crash)
           if proc.poll() is None:
               proc.terminate()
               print("  PASS: Executable launches without immediate crash")
               return True
           else:
               stdout, stderr = proc.communicate()
               print(f"  FAIL: Process exited early")
               print(f"  stdout: {stdout[-500:]}")  # Last 500 chars
               print(f"  stderr: {stderr[-500:]}")
               return False
       except Exception as e:
           print(f"  FAIL: Exception during launch: {e}")
           return False
   
   def test_internal_structure():
       """Verify _internal folder has expected structure."""
       print("\nTEST: Internal Structure")
       checks = [
           ("app/models", "Model directory"),
           ("app/assets", "Assets directory"),
           ("lesson/front_gif", "Front GIFs"),
           ("lesson/left_gif", "Left GIFs"),
           ("lesson/right_gif", "Right GIFs"),
           ("torch", "PyTorch libraries"),
           ("mediapipe", "MediaPipe libraries"),
           ("ultralytics", "Ultralytics libraries"),
       ]
       
       all_pass = True
       for subdir, description in checks:
           path = INTERNAL_PATH / subdir
           exists = path.exists()
           status = "PASS" if exists else "FAIL"
           print(f"  [{status}] {description}: {subdir}")
           if not exists:
               all_pass = False
       
       return all_pass
   
   def test_model_files():
       """Verify GCN model files are bundled."""
       print("\nTEST: Model Files")
       models = [
           "app/models/hybrid_gcn_v2_front.pth",
           "app/models/hybrid_gcn_v2_left.pth",
           "app/models/hybrid_gcn_v2_right.pth",
           "app/models/weights/best.pt",
           "yolov8n.pt",
       ]
       
       all_pass = True
       for model in models:
           path = INTERNAL_PATH / model
           exists = path.exists()
           size_mb = path.stat().st_size / (1024*1024) if exists else 0
           status = "PASS" if exists else "FAIL"
           print(f"  [{status}] {model} ({size_mb:.1f} MB)" if exists else f"  [{status}] {model} MISSING")
           if not exists:
               all_pass = False
       
       return all_pass
   
   def test_imports_in_packaged():
       """Test that key imports work in packaged environment."""
       print("\nTEST: Import Verification (via Python in _internal)")
       # This requires running python from within the packaged environment
       # or testing via a subprocess that uses the packaged python
       
       # For now, we'll check that the Python DLL/libs exist
       python_libs = list(INTERNAL_PATH.glob("python*.dll")) + list(INTERNAL_PATH.glob("libpython*.so"))
       if python_libs:
           print(f"  [PASS] Python runtime found: {python_libs[0].name}")
           return True
       else:
           print("  [WARN] Could not find Python runtime DLL (may be embedded)")
           return True  # Not a failure - PyInstaller may embed differently
   
   def main():
       print("="*60)
       print("TuroArnis Packaged Executable Tests")
       print("="*60)
       
       if not EXE_PATH.exists():
           print(f"\nERROR: {EXE_PATH} not found.")
           print("Run Plan 01 (Build) first.")
           return 1
       
       results = []
       results.append(("Launch", test_executable_launch()))
       results.append(("Structure", test_internal_structure()))
       results.append(("Models", test_model_files()))
       results.append(("Imports", test_imports_in_packaged()))
       
       print("\n" + "="*60)
       print("SUMMARY")
       print("="*60)
       all_pass = True
       for name, passed in results:
           status = "PASS" if passed else "FAIL"
           print(f"  [{status}] {name}")
           if not passed:
               all_pass = False
       
       return 0 if all_pass else 1
   
   if __name__ == "__main__":
       sys.exit(main())
   ```
